# Phase 2 — Authentication, demo-gating, console connect — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every non-public endpoint needs a scoped API key unless the deployment is an explicit demo; demo-only surfaces disappear outside demo; the console can hold a key and the review queue records who acted.

**Architecture:** `kavach/auth.py` owns the `api_keys` table (migration 2), key minting/hashing/lookup and a scope lattice. `apps/api_server.py` gets one dependency `require(scope)` that resolves the bearer key, and one `demo_only` dependency for lab surfaces. `KAVACH_AUTH` is `required` unless `KAVACH_DEMO=1`. The console keeps the key in `localStorage`, sends it as `Authorization: Bearer`, and routes a 401 to the Settings page's Connect card. A `/dashboard/access` page manages keys (operator scope).

**Tech Stack:** FastAPI dependencies, stdlib `secrets`/`hashlib`, Next 16 client components.

**Spec:** `docs/superpowers/specs/2026-09-13-production-readiness-design.md` §2

## Global Constraints

- No new dependency. Keys are `sha256`-hashed; only the hash is stored.
- Scopes: `readonly < agent < operator`. Key format `kv_<scope>_<32 url-safe chars>`.
- `KAVACH_DEMO` default in the image flips to `0`. Railway/compose set `1` explicitly already.
- Public (no key ever): `/api/health`, `/api/webhooks/*`, `/api/stepup/{token}` GET+resolve, `/api/checkout/{order_id}` GET, `/api/checkout/confirm`, `/api/checkout/{order_id}/link`, `/api/metrics` (unless `KAVACH_METRICS_KEY` is set, then it must match `?key=` or bearer).
- Demo-only (404 `demo_disabled` unless `KAVACH_DEMO=1`): `/api/storefront*`, `/api/duel`, `/api/scenarios*`, `/api/proof/tamper`, `/api/demo/*`, `/api/mcp/*`, `/api/checkout` POST (start), `/api/checkout/latest`.
- Every task: tests + ruff + commit.

```python
"""API keys: who is calling, and what they may do.

Three scopes, ordered: readonly < agent < operator. A key is minted once, shown once, and
stored only as a SHA-256 hash -- the ledger of keys must not itself be a credential store
worth stealing. Verification hashes the presented token and looks it up; there is no
timing side channel to speak of because the hash is the index, not a comparison.

KAVACH_AUTH decides whether the API demands a key. It defaults to `required`, except that a
deployment which explicitly says KAVACH_DEMO=1 defaults to `off` -- a demo a judge opens
from a link cannot ask them for a key. Either default can be overridden.
"""

from __future__ import annotations

import hashlib
import os
import secrets
from dataclasses import dataclass

from . import db

SCHEMA = """
CREATE TABLE IF NOT EXISTS api_keys (
    key_id       TEXT PRIMARY KEY,
    key_hash     TEXT NOT NULL UNIQUE,
    name         TEXT NOT NULL,
    scope        TEXT NOT NULL,
    created_at   INTEGER NOT NULL,
    revoked_at   INTEGER,
    last_used_at INTEGER
);
"""

SCOPES: tuple[str, ...] = ("readonly", "agent", "operator")


@dataclass(frozen=True)
class Key:
    key_id: str
    name: str
    scope: str


def allows(held: str, needed: str) -> bool:
    return SCOPES.index(held) >= SCOPES.index(needed)


def mode() -> str:
    explicit = os.environ.get("KAVACH_AUTH", "").strip().lower()
    if explicit in {"required", "off"}:
        return explicit
    demo = os.environ.get("KAVACH_DEMO", "").strip().lower() in {"1", "true", "on"}
    return "off" if demo else "required"


def _hash(token: str) -> str:
    return hashlib.sha256(token.encode()).hexdigest()


def create(conn: db.Connection, *, name: str, scope: str, now: int) -> dict:
    if scope not in SCOPES:
        raise ValueError(f"scope must be one of {', '.join(SCOPES)}")
    if not name or len(name) > 64:
        raise ValueError("name must be 1-64 characters")
    token = f"kv_{scope}_{secrets.token_urlsafe(24)}"
    key_id = f"key_{secrets.token_hex(6)}"
    conn.execute("INSERT INTO api_keys (key_id, key_hash, name, scope, created_at) "
                 "VALUES (?,?,?,?,?)", (key_id, _hash(token), name, scope, now))
    return {"key": token, "key_id": key_id, "name": name, "scope": scope, "created_at": now}


def verify(conn: db.Connection, token: str, *, now: int) -> Key | None:
    if not token or not token.startswith("kv_"):
        return None
    row = conn.execute("SELECT key_id, name, scope FROM api_keys WHERE key_hash=? "
                       "AND revoked_at IS NULL", (_hash(token),)).fetchone()
    if row is None:
        return None
    conn.execute("UPDATE api_keys SET last_used_at=? WHERE key_id=?", (now, row["key_id"]))
    return Key(row["key_id"], row["name"], row["scope"])


def listing(conn: db.Connection) -> list[dict]:
    rows = conn.execute("SELECT key_id, name, scope, created_at, revoked_at, last_used_at "
                        "FROM api_keys ORDER BY created_at DESC").fetchall()
    return [dict(r) for r in rows]


def revoke(conn: db.Connection, key_id: str, *, now: int) -> bool:
    cur = conn.execute("UPDATE api_keys SET revoked_at=? WHERE key_id=? AND revoked_at IS NULL",
                       (now, key_id))
    return cur.rowcount == 1
```

```python
"""`python -m kavach`: operator commands that need no running server."""

from __future__ import annotations

import argparse
import json
import os
import sys
import time

from . import auth, migrations
from .eventlog import connect


def _conn(path: str):
    conn = connect(path)
    migrations.apply(conn)
    return conn


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(prog="kavach")
    p.add_argument("--db", default=os.environ.get("KAVACH_DB", "kavach.db"),
                   help="SQLite path or postgresql:// URL (default: $KAVACH_DB)")
    sub = p.add_subparsers(dest="cmd", required=True)

    keys = sub.add_parser("keys", help="API keys").add_subparsers(dest="op", required=True)
    c = keys.add_parser("create")
    c.add_argument("--name", required=True)
    c.add_argument("--scope", required=True, choices=auth.SCOPES)
    keys.add_parser("list")
    r = keys.add_parser("revoke")
    r.add_argument("key_id")

    a = p.parse_args(argv)
    conn = _conn(a.db)
    try:
        if a.cmd == "keys" and a.op == "create":
            out = auth.create(conn, name=a.name, scope=a.scope, now=int(time.time()))
            print(json.dumps(out, indent=2))
            print("store this key now; it is not shown again", file=sys.stderr)
        elif a.cmd == "keys" and a.op == "list":
            print(json.dumps(auth.listing(conn), indent=2))
        elif a.cmd == "keys" and a.op == "revoke":
            ok = auth.revoke(conn, a.key_id, now=int(time.time()))
            print(json.dumps({"revoked": ok, "key_id": a.key_id}))
            return 0 if ok else 1
    finally:
        conn.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# ------------------------------------------------------------------ authentication

_BEARER = re.compile(r"^Bearer\s+(\S+)$", re.I)


def caller(request: Request, conn: Conn) -> auth.Key | None:
    """The key behind this request, or None when KAVACH_AUTH is off. Never raises: a route
    that needs a scope says so with require()."""
    header = request.headers.get("authorization", "")
    m = _BEARER.match(header)
    if not m:
        return None
    key = auth.verify(conn, m.group(1), now=int(time.time()))
    request.state.key = key
    return key


Caller = Annotated[auth.Key | None, Depends(caller)]


def require(scope: str):
    """A dependency that admits only a key holding at least `scope`. When auth is off it
    admits everyone, and reports the caller as the anonymous demo operator."""
    def dep(request: Request, key: Caller) -> auth.Key:
        if auth.mode() == "off":
            return key or auth.Key("anonymous", "operator", "operator")
        if key is None:
            code = "invalid_key" if request.headers.get("authorization") else "missing_key"
            raise _fail(401, code, "this endpoint needs an API key: send "
                                   "`Authorization: Bearer kv_...`; mint one with "
                                   "`python -m kavach keys create`")
        if not auth.allows(key.scope, scope):
            raise _fail(403, "insufficient_scope",
                        f"this key holds the {key.scope} scope; {scope} is needed")
        return key
    return dep


ReadOnly = Annotated[auth.Key, Depends(require("readonly"))]
Agent = Annotated[auth.Key, Depends(require("agent"))]
Operator = Annotated[auth.Key, Depends(require("operator"))]


def demo_only() -> None:
    """Lab and storefront surfaces exist only where KAVACH_DEMO=1 says so. A 404, not a
    403: outside a demo these routes are not a thing this deployment has."""
    if not demo.enabled():
        raise _fail(404, "demo_disabled",
                    "this surface is only mounted when KAVACH_DEMO=1")


Demo = Annotated[None, Depends(demo_only)]
```

class KeyCreateRequest(BaseModel):
    name: str = Field(min_length=1, max_length=64, pattern=r"^[A-Za-z0-9 ._-]+$")
    scope: Literal["readonly", "agent", "operator"]

@app.post("/api/keys", status_code=201)
def keys_create(body: KeyCreateRequest, conn: Conn, _: Operator) -> dict[str, Any]:
    return {**auth.create(conn, name=body.name, scope=body.scope, now=int(time.time())),
            "note": "store the key now; it is not shown again"}


@app.delete("/api/keys/{key_id}")
def keys_revoke(key_id: str, conn: Conn, _: Operator) -> dict[str, Any]:
    if not re.match(r"^key_[a-f0-9]{12}$", key_id):

    if not auth.revoke(conn, key_id, now=int(time.time())):
        raise _fail(404, "key_not_found", "no such active key")
    return {"revoked": True, "key_id": key_id}

   an explicit CORS origin, and a bearer header cannot be sent by a cross-site form. */
const KEY_STORAGE = 'kavach.api_key';

export function getApiKey(): string {
  try { return localStorage.getItem(KEY_STORAGE) ?? ''; } catch { return ''; }
}
export function setApiKey(key: string): void {
  try { key ? localStorage.setItem(KEY_STORAGE, key) : localStorage.removeItem(KEY_STORAGE); }
  catch { /* private mode: the key lives for this page load only */ }
  memoryKey = key;
}
let memoryKey = '';
function authHeader(): Record<string, string> {
  const k = memoryKey || getApiKey();
  return k ? { Authorization: `Bearer ${k}` } : {};
}
